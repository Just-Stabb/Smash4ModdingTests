# Variable names for tooltips: 0xVARIABLE = Name | description (\n = new line).
# Hover a variable in a script to see it. Add your own lines freely.
#
# Common position variables (tools/position_vars_cave.py, any fighter). Use them with SET_FLOAT
# (write) and Float_Var_Compare (read); "fwd" = X along this fighter's facing, positive = in front.
# "Opponent" = the nearest other fighter. Reads give 0 when there is no such fighter.
0x0F000000 = Self X | Read: this fighter's X.\nWrite: set it.
0x0F000001 = Self Y | Read: this fighter's Y.\nWrite: set it.
0x0F000002 = Self Z | Read: this fighter's Z.\nWrite: set it.
0x0F000003 = Self facing | Read: 1 facing right, -1 facing left. Writes are ignored.
0x0F000004 = Move self forward | Write: move this fighter forward (fwd) by the value. Reads 0.
0x0F000005 = Move self up | Write: move this fighter up by the value. Reads 0.
0x0F000006 = Move self Z | Write: move this fighter along Z by the value. Reads 0.
0x0F000008 = Opponent X | Read: the nearest opponent's X.\nWrite: set it.
0x0F000009 = Opponent Y | Read: the nearest opponent's Y.\nWrite: set it.
0x0F00000A = Opponent Z | Read: the nearest opponent's Z.\nWrite: set it.
0x0F00000B = Opponent distance | Read: X/Y distance to the nearest opponent. Writes are ignored.
0x0F00000C = Opponent offset X (fwd) | Read: opponent X - self X, along self's facing (positive = opponent in front).\nWrite: put the opponent that far in front of this fighter.
0x0F00000D = Opponent offset Y | Read: opponent Y - self Y.\nWrite: put the opponent at self Y + value.
0x0F00000E = Opponent offset Z | Read: opponent Z - self Z.\nWrite: put the opponent at self Z + value.
0x0F000010 = Self offset X (fwd) | Read: self X - opponent X, along self's facing.\nWrite: put this fighter at opponent X + value (fwd); negative = on the side self faces from.
0x0F000011 = Self offset Y | Read: self Y - opponent Y.\nWrite: put this fighter at opponent Y + value.
0x0F000012 = Self offset Z | Read: self Z - opponent Z.\nWrite: put this fighter at opponent Z + value.
0x0F000020 = Slot 1 X | Player slot 1 (fighter table entry 0). Read / write.
0x0F000021 = Slot 1 Y | Player slot 1. Read / write.
0x0F000022 = Slot 1 Z | Player slot 1. Read / write.
0x0F000023 = Slot 1 facing | Player slot 1: 1 right, -1 left. Read only.
0x0F000024 = Slot 2 X | Player slot 2. Read / write.
0x0F000025 = Slot 2 Y | Player slot 2. Read / write.
0x0F000026 = Slot 2 Z | Player slot 2. Read / write.
0x0F000027 = Slot 2 facing | Player slot 2. Read only.
0x0F000028 = Slot 3 X | Player slot 3. Read / write.
0x0F000029 = Slot 3 Y | Player slot 3. Read / write.
0x0F00002A = Slot 3 Z | Player slot 3. Read / write.
0x0F00002B = Slot 3 facing | Player slot 3. Read only.
0x0F00002C = Slot 4 X | Player slot 4. Read / write.
0x0F00002D = Slot 4 Y | Player slot 4. Read / write.
0x0F00002E = Slot 4 Z | Player slot 4. Read / write.
0x0F00002F = Slot 4 facing | Player slot 4. Read only.
0x0F000030 = Slot 5 X | Player slot 5. Read / write.
0x0F000031 = Slot 5 Y | Player slot 5. Read / write.
0x0F000032 = Slot 5 Z | Player slot 5. Read / write.
0x0F000033 = Slot 5 facing | Player slot 5. Read only.
0x0F000034 = Slot 6 X | Player slot 6. Read / write.
0x0F000035 = Slot 6 Y | Player slot 6. Read / write.
0x0F000036 = Slot 6 Z | Player slot 6. Read / write.
0x0F000037 = Slot 6 facing | Player slot 6. Read only.
0x0F000038 = Slot 7 X | Player slot 7. Read / write.
0x0F000039 = Slot 7 Y | Player slot 7. Read / write.
0x0F00003A = Slot 7 Z | Player slot 7. Read / write.
0x0F00003B = Slot 7 facing | Player slot 7. Read only.
0x0F00003C = Slot 8 X | Player slot 8. Read / write.
0x0F00003D = Slot 8 Y | Player slot 8. Read / write.
